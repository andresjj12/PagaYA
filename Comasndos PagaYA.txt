========================================================

5. Comprobar las relaciones

(Billeteras de los usuarios)
SELECT
    b.id AS Billetera,
    b.usuario_id,
    u.nombre,
    u.apellido,
    u.email,
    b.saldo,
    b.moneda,
    b.estado
FROM PagaYA_SQL.dbo.Billeteras b
INNER JOIN PagaYA_SQL.dbo.Usuarios u
    ON b.usuario_id = u.id
ORDER BY b.id;

(Transacciones de los usuarios)
SELECT
    t.id AS Transaccion,
    t.usuario_id,
    u.email,
    t.billetera_id,
    t.tipo,
    t.monto,
    t.saldo_anterior,
    t.saldo_nuevo,
    t.estado,
    t.referencia
FROM PagaYA_SQL.dbo.Transacciones t
INNER JOIN PagaYA_SQL.dbo.Usuarios u
    ON t.usuario_id = u.id
INNER JOIN PagaYA_SQL.dbo.Billeteras b
    ON t.billetera_id = b.id
ORDER BY t.id;

6. Limpiar el Backup antes de copiar

USE PagaYA_Backup;
GO

DELETE FROM dbo.Transacciones;
DELETE FROM dbo.Billeteras;
DELETE FROM dbo.Usuarios;
GO
